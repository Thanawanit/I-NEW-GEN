Software Architecture, Mathematical Formulas & Crop Database Spec — `v1.4`

> วัตถุประสงค์เอกสาร: กำหนดสถาปัตยกรรมซอฟต์แวร์ โครงสร้างฐานข้อมูลพืช/พันธุ์/ระยะการเจริญเติบโต ชุดสูตรคำนวณทางคณิตศาสตร์ทั้งหมด (พร้อมตัวอย่างการแทนค่า) และ API Contract สำหรับเชื่อมต่อเว็บ, โดรน และบอร์ด ESP32

1. สถาปัตยกรรมซอฟต์แวร์ (Software Stack)

| เลเยอร์ | เทคโนโลยีที่กำหนด | หน้าที่หลัก |
| :--- | :--- | :--- |
| Frontend (Map & UI) | HTML5 + CSS + JavaScript + Leaflet.js + Leaflet.draw (`L.GeometryUtil.geodesicArea`) + Esri World Imagery / OpenStreetMap | วาด Polygon ขอบเขตแปลงบนแผนที่ภาพถ่ายดาวเทียมฟรี (ไม่ต้องใช้บัตรเครดิต/API Key), คำนวณพื้นที่ (m^2) แบบ Geodesic พร้อมช่องให้ผู้ใช้แก้ไขเลขเอง, ฟอร์มเลือกพืช/พันธุ์/ระยะเติบโต, และ Dashboard แสดงผลเรียลไทม์ |
| Backend Server | Python Flask (`host='0.0.0.0'`, พอร์ต `5000`) | คำนวณสูตรปุ๋ยและระยะบิน, ประมวลผลภาพถ่ายโดรน, สื่อสารกับโดรนผ่าน USB Wi-Fi, และเปิด REST API ให้ ESP32 ส่งค่า/รับคำสั่ง |
| Image Processing | OpenCV (`cv2`) + NumPy | อ่านภาพถ่ายจากโดรนรอบที่ 1 (OpenCV อ่านสีเป็น BGR ต้องแปลงเป็น `int16` ก่อนคำนวณ 2G - R - B เพื่อป้องกัน `uint8` overflow) |
| Database | SQLite | จัดเก็บข้อมูลแปลงเกษตร, เกณฑ์ธาตุอาหารพืช 3 ชนิด, ค่าตรวจดินสด, สถานะถังหมัก 3 ถัง และประวัติการสั่งผสม/บินพ่น |
| IoT Firmware | Arduino C++ (สำหรับ ESP32 #1 ชุดดิน และ ESP32 #2 เครื่องหมัก/ผสม) | อ่านค่าเซนเซอร์และยิง HTTP POST/GET JSON เข้า Flask Server ผ่าน Wi-Fi วงเดียวกับโน้ตบุ๊ก |

2. โครงสร้างฐานข้อมูล (SQLite Schema)

-- 1. ตารางเกณฑ์ธาตุอาหารอ้างอิงตาม พืช + พันธุ์ + ระยะการเจริญเติบโต
CREATE TABLE crop_profiles (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    crop_name TEXT NOT NULL,          -- เช่น 'ข้าวโพดเลี้ยงสัตว์'
    variety TEXT NOT NULL,            -- เช่น 'ลูกผสมเดี่ยว (ทั่วไป)'
    growth_stage TEXT NOT NULL,       -- 'seedling', 'vegetative', 'reproductive'
    root_depth_m REAL NOT NULL,       -- ความลึกรากอ้างอิง (เมตร)
    target_n_g_m2 REAL NOT NULL,      -- ความต้องการ N เป้าหมาย (g/m^2)
    target_p2o5_g_m2 REAL NOT NULL,   -- ความต้องการ P2O5 เป้าหมาย (g/m^2)
    target_k2o_g_m2 REAL NOT NULL,    -- ความต้องการ K2O เป้าหมาย (g/m^2)
    ph_min REAL NOT NULL,             -- ช่วง pH เหมาะสมต่ำสุด
    ph_max REAL NOT NULL,             -- ช่วง pH เหมาะสมสูงสุด
    safe_ec_max_ppm REAL NOT NULL     -- ค่า EC/TDS สูงสุดที่ปลอดภัยในการพ่นทางใบ (ppm)
);

-- 2. ตารางแปลงเกษตรที่ผู้ใช้วาดและยืนยันพื้นที่
CREATE TABLE plots (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    plot_name TEXT NOT NULL,
    polygon_geojson TEXT NOT NULL,    -- พิกัด Lat/Lng ทั้งหมดจาก Leaflet
    calculated_area_m2 REAL NOT NULL, -- พื้นที่ที่แผนที่คำนวณได้
    confirmed_area_m2 REAL NOT NULL,  -- พื้นที่ที่ผู้ใช้กดยืนยันหรือแก้ไขเอง
    soil_type TEXT NOT NULL,          -- เช่น 'sandy_loam', 'clay_loam'
    bulk_density_g_cm3 REAL NOT NULL, -- ความหนาแน่นมวลดิน (g/cm^3)
    crop_profile_id INTEGER NOT NULL,
    mode TEXT NOT NULL DEFAULT 'live',-- 'live' หรือ 'sim' (แยกฐานข้อมูล 100%)
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (crop_profile_id) REFERENCES crop_profiles(id)
);

-- 3. ตารางบันทึกค่าดิน (จาก ESP32 #1 + ชุดตรวจดิน; ระบบรวมค่าความชื้นล่าสุดกับค่า NPK ล่าสุดของ plot_id เดียวกันให้อัตโนมัติ)
CREATE TABLE soil_readings (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    plot_id INTEGER NOT NULL,
    nitrogen_mg_kg REAL,
    phosphorus_mg_kg REAL,
    potassium_mg_kg REAL,
    ph_value REAL,
    moisture_pct REAL,
    ec_us_cm REAL,
    source TEXT NOT NULL,             -- 'esp32_live', 'manual_testkit', หรือ 'simulated'
    mode TEXT NOT NULL DEFAULT 'live',-- 'live' หรือ 'sim' (แยกฐานข้อมูล 100%)
    recorded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 4. ตารางสถานะถังหมักชีวภาพ 3 ถัง (จาก ESP32 #2)
CREATE TABLE fermenter_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    temp_n_c REAL,
    temp_p_c REAL,
    temp_k_c REAL,
    temp_mixer_c REAL,                -- คอลัมน์ใน SQLite (เมื่อส่งออก JSON API จะแปลงชื่อเป็น temp_n, temp_p, temp_k, temp_mixer)
    mixer_tds_ppm REAL,
    is_ready INTEGER DEFAULT 0,       -- 1 เมื่ออุณหภูมิ 3 ถังอยู่ในช่วง 20–45 °C
    source TEXT NOT NULL DEFAULT 'esp32_live', -- 'esp32_live' หรือ 'simulated'
    mode TEXT NOT NULL DEFAULT 'live',
    recorded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 5. ตารางคำสั่งผสมปุ๋ยและประวัติการบิน
CREATE TABLE mixing_jobs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    plot_id INTEGER NOT NULL,
    delta_n_g REAL NOT NULL,          -- ปริมาณขาด N (g)
    delta_p_g REAL NOT NULL,          -- ปริมาณขาด P2O5 (g)
    delta_k_g REAL NOT NULL,          -- ปริมาณขาด K2O (g)
    vol_n_ml REAL NOT NULL,
    vol_p_ml REAL NOT NULL,
    vol_k_ml REAL NOT NULL,
    vol_water_ml REAL NOT NULL,
    pump_n_sec REAL NOT NULL,         -- เวลาเปิดปั๊ม (วินาที; API ส่ง ms = round(sec*1000))
    pump_p_sec REAL NOT NULL,
    pump_k_sec REAL NOT NULL,
    pump_water_sec REAL NOT NULL,
    final_tds_ppm REAL,
    status TEXT NOT NULL,             -- 'pending', 'mixing', 'diluting', 'ready', 'sprayed'
    mode TEXT NOT NULL DEFAULT 'live',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

4. ชุดสูตรคำนวณคณิตศาสตร์ทั้งหมด (พร้อมตัวอย่างแทนค่า)

สูตรที่ 1: การคำนวณพื้นที่ Polygon และการยืนยันค่า
- บน Frontend ใช้ `L.GeometryUtil.geodesicArea(latlngs)` คำนวณพื้นที่ผิวโลก (หน่วย m^2) จากจุดมุม Polygon
- แสดงค่า `calculated_area_m2` ในช่อง Input ที่แก้ไขได้ -> เก็บค่าจริงที่ผู้ใช้ยืนยันเป็น `confirmed_area_m2` (A_{real})

สูตรที่ 2: การย่อสเกลเส้นทางบินจากแผนที่ลงสู่ถาดแปลงบนโต๊ะ (Scale Calibration)
กำหนดกรอบพื้นที่บินปลอดภัยบนถาดแปลง `60 × 80 cm` (หักขอบปลอดภัยด้านละ 10 cm) ให้เหลือพื้นที่บินสุทธิ W_{table} = 40 cm และความยาว H_{table} = 60 cm

1. แปลงพิกัด Lat/Lng (องศา) เป็นพิกัดเมตร (x_i, y_i) เทียบกับจุดมุมแรก (lat_0, lng_0) ด้วย Equirectangular Projection (R_{earth} = 6{,}371{,}000 m, แปลงองศาเป็นเรเดียนก่อนคูณ):
   x_i (m) = R_{earth} * (lng_i - lng_0) * \cos(lat_0), \qquad y_i (m) = R_{earth} * (lat_i - lat_0)
2. หา Bounding Box ของแปลงบนแผนที่จริง: ความกว้าง W_{real} = x_{max} - x_{min} (m) และความยาว H_{real} = y_{max} - y_{min} (m)
3. หาตัวคูณย่อสเกล k (หน่วย cm/m):
   k = min(\frac{W_{table}}{W_{real}},\ \frac{H_{table}}{H_{real}})
4. แปลงจุดมุมแปลงแต่ละจุดเป็นคำสั่งเคลื่อนที่ของโดรน (กำหนดจุดเริ่มต้นที่มุมแปลงแรก และหันหัวโดรนไปทางทิศเหนือ `+Y` ของโต๊ะเป็น 0^ deg):
   Delta  X_i (cm) = k * (x_i - x_{i-1}), \qquad Delta  Y_i (cm) = k * (y_i - y_{i-1})
   ระยะบินตรง  d_i = sqrt{Delta  X_i^2 + Delta  Y_i^2} (cm), \qquad มุมทิศ  heading_i = atan2(Delta  X_i, Delta  Y_i)
   มุมสั่งเลี้ยว  turn_i = ((heading_i - heading_{i-1} + 180^ deg) mod 360^ deg) - 180^ deg
   *(ตัวอย่าง: ถ้า Delta  X=40, Delta  Y=0 จะได้ heading = 90^ deg คือเลี้ยวขวา 90^ deg จากทิศเหนือ และหากระยะ d_i < 20 cm ให้รวมช่วงนี้เข้ากับจุดถัดไปก่อนสั่งบิน โดย GeoJSON Polygon จะมีจุดสุดท้ายซ้ำกับจุดแรกเสมอทำให้บินครบรอบกลับมาจุดเริ่มต้นอัตโนมัติ)*

สูตรที่ 3: วิเคราะห์ภาพถ่ายโดรนด้วย Excess Green Index (ExG)
*(หมายเหตุสำหรับโค้ด: OpenCV อ่านภาพเป็น BGR ไม่ใช่ RGB และต้องแปลงอาร์เรย์เป็น `int16` หรือ `float` ก่อนคำนวณ เพื่อป้องกัน `uint8` overflow)*
สำหรับแต่ละพิกเซล (R, G, B) ในภาพถ่ายแปลงจำลอง (ช่วงค่า 0–255):
ExG(x, y) = 2G(x, y) - R(x, y) - B(x, y)
นับจำนวนพิกเซลที่ ExG(x, y) > T_{green} (เช่น T_{green} = 20 ปรับค่าตามแสงห้อง):
Green Coverage (\%) = \frac{จำนวนพิกเซลที่  ExG > T_{green}}{จำนวนพิกเซลทั้งหมดในกรอบแปลง} x 100

สูตรที่ 4: คำนวณปริมาณธาตุอาหารที่ดินขาด (Delta  N, Delta  P_2O_5, Delta  K_2O) และแจ้งเตือน pH

1. แปลงค่าธาตุอาหารที่มีอยู่ในดิน จากหน่วย mg/kg (ppm) เป็นมวลต่อพื้นที่ (g/m^2):
   มวลเนื้อดินต่อ  1 m^2 (kg/m^2) = rho_{soil}\ (g/cm^3) x d_{root}\ (m) x 1000
   M_{soil\_elem, i}\ (g/m^2) = C_{measured, i}\ (mg/kg) x (rho_{soil} x d_{root})
   *(หมายเหตุ: mg/kg x 1000\ kg/m^2 x 10^{-3}\ g/mg = mg/kg x 1 พอดี)*

2. แปลงรูปธาตุบริสุทธิ์เป็นรูปปุ๋ยออกไซด์และคูณสัมประสิทธิ์การดูดใช้ได้ต่อฤดู (f_N = 0.30,\ f_P = 0.20,\ f_K = 0.15 ซึ่งเป็นค่าพารามิเตอร์สมมติสำหรับโมเดลสาธิตเพื่อสะท้อนสัดส่วนที่พืชดูดใช้ได้จริงจากดินใน 1 ระยะ):
   M_{soil, N} = M_{soil\_elem, N} x 0.30, \qquad M_{soil, P_2O_5} = 2.29 x M_{soil\_elem, P} x 0.20, \qquad M_{soil, K_2O} = 1.20 x M_{soil\_elem, K} x 0.15

   > ตัวอย่างคำนวณ: ดินร่วน rho = 1.35\ g/cm^3, ข้าวโพดระยะเติบโต d_{root} = 0.25\ m, วัดค่า N ระดับกลางจาก LDD ได้ 15\ mg/kg และ P ระดับต่ำ 5\ mg/kg *(สมมติให้ความเข้มข้นเฉลี่ยในเขตรากพืชเท่ากับหน้าดินที่เก็บ 0–15 ซม.)*:
   > M_{soil, N} = 15 x 1.35 x 0.25 x 0.30 = 1.519\ g/m^2
   > M_{soil, P_2O_5} = 2.29 x (5 x 1.35 x 0.25) x 0.20 = 0.773\ g/m^2

3. คำนวณปริมาณธาตุที่ขาดทั้งแปลง (Delta  M_i หน่วย กรัม):
   Delta  M_i\ (g) = max(0,\ M_{target, i} - M_{soil, i}) x A_{real}\ (m^2)
   > ต่อจากตัวอย่าง: เป้าหมายข้าวโพดระยะเติบโต M_{target, N} = 6.00\ g/m^2, M_{target, P_2O_5} = 1.50\ g/m^2, พื้นที่จริง A_{real} = 100\ m^2:
   > Delta  M_{N} = (6.00 - 1.519) x 100 = 448.1\ g, \qquad Delta  M_{P_2O_5} = (1.50 - 0.773) x 100 = 72.7\ g
   *(กรณีใช้ LDD Soil Test Kit ที่ให้ผลเป็นระดับ `ต่ำ / ปานกลาง / สูง` ให้ตั้งค่าตัวแทน C_{measured} ของแต่ละระดับไว้ใน Config เพื่อเข้าสูตรเดียวกันได้ทันที)*

4. เงื่อนไขตรวจสอบ pH (Safety & Absorption Warning):
   - ถ้า pH < ph\_min -> แจ้งเตือน `"ดินเป็นกรดเกินเกณฑ์ พืชดูดซึมธาตุได้น้อย ควรปรับสภาพด้วยปูนขาว/โดโลไมต์ก่อนใส่ปุ๋ย"`
   - ถ้า pH > ph\_max -> แจ้งเตือน `"ดินเป็นด่างเกินเกณฑ์ ควรปรับสภาพด้วยอินทรียวัตถุ/กำมะถัน"`

สูตรที่ 5: คำนวณปริมาตรน้ำหมัก, ย่อสเกล Demo และเวลาเปิดปั๊ม (Pump Runtime)

1. ปริมาตรน้ำหมักที่ต้องการจริงตามทฤษฎี (เมื่อความเข้มข้นน้ำหมักแต่ละถังคือ C_{bio, i} หน่วย mg/ml ของ N, P_2O_5, K_2O จากค่า Placeholder ใน `config.py` — ตั้งเท่ากัน `2.0 mg/ml` ทั้ง 3 ถังเพื่อรักษาสัดส่วน Delta  M จนกว่าคนที่ 4 จะวัดค่าจริง):
   V_{real, i}\ (ml) = \frac{Delta  M_i\ (g) x 1000}{C_{bio, i}\ (mg/ml)}
2. การย่อสเกลปริมาตรสำหรับผสมโชว์บนโต๊ะ (Demo Batch Scaling):
   ผสมในถังกลางปริมาตรรวม V_{batch} = 200 ml โดยล็อกปริมาณน้ำหมักรวมคงที่ V_{demo\_bio} = 80 ml และใช้ Delta  M กำหนด อัตราส่วน N : P : K (ส่วนที่เหลือ 120 ml เป็นน้ำเปล่าเจือจาง):
   V_{demo, i} = V_{demo\_bio} x \frac{V_{real, i}}{V_{real, N} + V_{real, P} + V_{real, K}}
   *(กรณีที่ดินไม่ขาดธาตุใดเลย คือผลรวม V_{real} = 0 ให้บันทึก `status = 'ready'`, คืนค่าเวลาทุกปั๊มเป็น `0.0 s` และแจ้งเตือนบนเว็บว่า "ดินมีความสมบูรณ์เพียงพอ ไม่ต้องเติมปุ๋ย")*

3. คำนวณเวลาเปิด-ปิดปั๊ม 4 ตัว (t_{pump, i}) พร้อม Minimum Runtime Guard:
   t_{pump, i}\ (วินาที) = \frac{V_{demo, i}\ (ml)}{Q_{pump, i}\ (ml/s)}
   - กติกาเดียวป้องกัน Pump Dead-Time: ก่อนเริ่มสั่งงานต้องไล่น้ำเต็มสายยางทุกเส้น (Pre-priming) และหากปั๊มธาตุใดคำนวณเวลาได้ 0 < t_{pump, i} < 1.5 s ให้ ตัดเป็น `0.0 s` แล้วย้ายปริมาตรส่วนนั้นไปบวกในน้ำเปล่าเพื่อให้รวมยังเป็น 200 ml โดยเก็บ `pump_seconds` ปัดทศนิยม 2 ตำแหน่ง (`round(sec, 2)`) ลงฐานข้อมูล แล้วแปลงเป็นมิลลิวินาทีใน API ด้วย `int(round(pump_sec * 1000))`
   > ตัวอย่างคำนวณ: ปั๊ม N มี Q = 5.0\ ml/s, ปริมาตร V_{demo, N} = 68.83 ml -> t_{pump, N} = round(68.8326 / 5.0, 2) = 13.77 s (`13770 ms`) และปั๊ม P มี V_{demo, P} = 11.17 ml -> t_{pump, P} = round(11.1674 / 5.0, 2) = 2.23 s (`2230 ms`)

4. ลูปตรวจเช็กความเค็มและเจือจาง TDS/EC กึ่งอัตโนมัติ (Salinity/Osmotic Burn Guard):
   หลัง ESP32 #2 เปิดปั๊มครบ 4 ตัว ให้ผู้ใช้กวนถังผสมแล้วจุ่มหัววัด TDS รอ 5 วินาที เพื่อให้ ESP32 #2 ส่งค่า `mixer_tds_ppm` (มัธยฐานจาก 30 ตัวอย่าง) เข้า `/api/fermenter/telemetry` แล้วกดปุ่มตรวจบนเว็บ (`POST /api/mixer/check-ec`):
   - *(หมายเหตุ: คอลัมน์ `safe_ec_max_ppm` เก็บเป็นค่า TDS หน่วย ppm โดย TDS ppm = EC uS/cm x 0.5 และห้ามนำค่านี้ไปเทียบตรงๆ กับ `ec_us_cm` ของดิน)*
   - ถ้า `mixer_tds_ppm` <= `safe_ec_max_ppm` -> สถานะ = `'ready'` (อนุญาตให้ตักขึ้นโดรนได้ และล้างหัววัดด้วยน้ำสะอาด)
   - ถ้า `mixer_tds_ppm` > `safe_ec_max_ppm` -> Flask คำนวณปริมาตรน้ำเปล่าเติมเพิ่มโดยคิดค่า TDS พื้นฐานของน้ำประปา (EC_{water} ~ 150\ ppm) และใช้ปริมาตรสะสมปัจจุบัน V_{current} = V_{N} + V_{P} + V_{K} + V_{water}:
     V_{add\_water} = V_{current} x \frac{EC_{current} - EC_{safe}}{EC_{safe} - EC_{water}}
     *(ตรวจเช็กความจุถัง: หาก V_{current} + V_{add\_water} > 500 ml ให้หยุดและแจ้งเตือนผู้ใช้แบ่งน้ำปุ๋ยออกครึ่งหนึ่งก่อน)* มิฉะนั้นระบบจะตั้ง `status = 'diluting'`, บวก `vol_water_ml` สะสมเพิ่ม, รีเซ็ตเวลาปั๊ม N/P/K เป็น `0`, และใส่เวลาปั๊มน้ำเปล่า t_{add} = V_{add\_water} / Q_{pump, water} ให้ ESP32 #2 ดึงไปเติมน้ำเจือจางแล้ววัดซ้ำ

6. ค่าคงที่อ้างอิงและ State Machine กลาง (ห้ามเปลี่ยนแปลงโดยพลการ)

6.1 ตารางแปลงผล LDD Soil Test Kit (`MANUAL`) เป็นค่าตัวแทนสมมติ `mg/kg`
*(หมายเหตุทางวิชาการ: ชุดตรวจดินภาคสนาม LDD ให้ผลเชิงกึ่งปริมาณ จึงกำหนดค่าตัวแทนสมมติเทียบเคียงช่วงเกณฑ์วิเคราะห์ดินของกรมพัฒนาที่ดินไว้ใน `config.py` เพื่อให้ทีมทุกคนคำนวณตรงกัน รองรับทั้งแบบ 3 ระดับและ 5 ระดับ)*:
- ไนโตรเจน (N): `very_low` = `2.5`, `low` = `5.0`, `medium` = `15.0`, `high` = `30.0`, `very_high` = `45.0 mg/kg`
- ฟอสฟอรัส (P — เทียบช่วง Bray II): `very_low` = `2.0`, `low` = `5.0`, `medium` = `15.0`, `high` = `30.0`, `very_high` = `50.0 mg/kg`
- โพแทสเซียม (K — เทียบช่วง NH₄OAc): `very_low` = `20.0`, `low` = `30.0`, `medium` = `60.0`, `high` = `100.0`, `very_high` = `130.0 mg/kg`

6.2 ตารางความหนาแน่นมวลดินอ้างอิง (rho_{soil})
- `sandy_loam` (ดินร่วนปนทราย) = `1.45 g/cm³`
- `loam` (ดินร่วน) = `1.35 g/cm³`
- `clay_loam` (ดินร่วนเหนียว) = `1.25 g/cm³`

6.3 State Machine ของคำสั่งผสมปุ๋ย (`mixing_jobs.status`)
1. `pending` — คำนวณสูตรเสร็จ รอ ESP32 #2 ดึงคำสั่งผ่าน `GET /api/mixer/command`
2. `mixing` — ESP32 #2 รับคำสั่งไปแล้ว กำลังเปิดปั๊ม N, P, K, น้ำเปล่า ทีละตัว
3. `diluting` — หลังผสมพบ `mixer_tds_ppm > safe_ec_max_ppm` กำลังสั่งปั๊มน้ำเปล่าเติมเจือจางเพิ่ม
4. `ready` — ผ่านเกณฑ์ EC ปลอดภัย พร้อมตักน้ำปุ๋ยใส่ตลับพ่นโดรน
5. `sprayed` — โดรนรอบที่ 2 ปฏิบัติภารกิจพ่นปุ๋ยเสร็จสิ้น
